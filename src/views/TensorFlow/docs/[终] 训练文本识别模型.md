```python
class OCRDataset(Dataset):
  def __init__(self, data_dir, transform=None):
    self.data_dir = data_dir
    self.transform = transform
    self.image_paths = []
    self.labels = []
    labels_path = os.path.join(data_dir, "labels.txt")
    with open(labels_path, "r", encoding="utf-8") as f:
		for line in f:
			path, label = line.strip().split("\t")
			self.image_paths.append(os.path.join(data_dir, path))
			self.labels.append(label)

  def __len__(self):
    return len(self.labels)

  def __getitem__(self, idx):
    img_path = self.image_paths[idx]
    label = self.labels[idx]
    image = Image.open(img_path).convert("L") # 直接转为灰度
    if self.transform:
		image = self.transform(image)

    label_tensor = torch.LongTensor([char_to_int[c] for c in label])
    return image, label_tensor
```

```python
EPOCHS = 200;
PATIENCE = 15;
best_val_loss = float('inf');
patience_counter = 0
MODEL_SAVE_PATH = "best_crnn_websafe_model.pth"

# --- 4. 【核心替换】定义全新的、Web友好的模型架构 ---
class ResNetBlock(nn.Module):
  def __init__(self, in_channels, out_channels, stride=1):
    super(ResNetBlock, self).__init__()
    self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=1, bias=False)
    self.bn1 = nn.BatchNorm2d(out_channels)
    self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=1, padding=1, bias=False)
    self.bn2 = nn.BatchNorm2d(out_channels)

    self.shortcut = nn.Sequential()
    if stride != 1 or in_channels != out_channels:
        self.shortcut = nn.Sequential(
        nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=stride, bias=False),
        nn.BatchNorm2d(out_channels)
        )

    def forward(self, x):
        identity = self.shortcut(x)
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += identity  # 简单的加法，ONNX 支持良好
        return F.relu(out)

# 2. 强大的 ResNet-CRNN 架构
class CRNN_ResNet_Final(nn.Module):
    def __init__(self, vocab_size):
        super(CRNN_ResNet_Final, self).__init__()
        self.cnn = nn.Sequential(
            nn.Conv2d(1, 64, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(64), nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2), # 64x32x128

            ResNetBlock(64, 128, stride=(2, 1)),   # 128x16x128
            ResNetBlock(128, 256, stride=(2, 1)),  # 256x8x128
            ResNetBlock(256, 512, stride=(2, 2)),  # 512x4x64
        )
        # 512 * 4 = 2048
        self.rnn = nn.LSTM(2048, 256, bidirectional=True, num_layers=2, batch_first=True, dropout=0.5)
        self.fc = nn.Linear(512, vocab_size)

    def forward(self, x):
        x = self.cnn(x)
        x = x.permute(0, 3, 1, 2) # [B, W, C, H]
        x = x.flatten(2)          # [B, W, C*H]
        x, _ = self.rnn(x)
        x = self.fc(x)
        return x.log_softmax(2)

# 实例化新模型
model = CRNN_ResNet_Final(VOCAB_SIZE).to(DEVICE)
optimizer = optim.Adam(model.parameters(), lr=0.001)
criterion = nn.CTCLoss(blank=0, zero_infinity=True)
# 添加一个学习率调度器，这里使用余弦退火
```

## 10. 训练模型

```python
# 定义解码与可视化函数
def decode_predictions(preds, int_to_char):
  texts = []
  preds = preds.argmax(2)
  for pred in preds:
    text = ""
    for i, token_index in enumerate(pred):
      token = int_to_char[token_index.item()]
      if token != "[BLANK]" and (i == 0 or token != int_to_char[pred[i-1].item()]):
        text += token
    texts.append(text)
  return texts
print("🚀🚀🚀 开始全新训练 (Web-Safe模型)... 🚀🚀🚀")
for epoch in range(EPOCHS):
    model.train()
    for images, labels, label_lengths in tqdm(train_loader, desc=f"Epoch {epoch+1}/{EPOCHS} [训练]"):
        images, labels, label_lengths = images.to(DEVICE), labels.to(DEVICE), label_lengths.to(DEVICE)

        optimizer.zero_grad();

        preds = model(images).permute(1, 0, 2)
        input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)
        loss = criterion(preds, labels, input_lengths, label_lengths);
        loss.backward();
        optimizer.step()
        # 在每个epoch的训练循环结束后
        scheduler.step()
    model.eval();
    val_loss = 0
    with torch.no_grad():
        for i, (images, labels, label_lengths) in enumerate(val_loader):
            images, labels, label_lengths = images.to(DEVICE), labels.to(DEVICE), label_lengths.to(DEVICE)
            preds = model(images).permute(1, 0, 2)
            input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)
            loss = criterion(preds, labels, input_lengths, label_lengths);
            val_loss += loss.item()

        if i == 0:
            pred_texts = decode_predictions(preds.permute(1, 0, 2), int_to_char)
            print(f"\n--- Epoch {epoch+1} 预测示例 ---")
            for j in range(min(5, len(pred_texts))):
                real_text = "".join([int_to_char.get(c.item(), '') for c in labels[j][:label_lengths[j]]]); print(f"  真实: {real_text:<20} -> 预测: {pred_texts[j]}")
            print("-" * 30)
    val_loss /= len(val_loader);
    print(f"Epoch {epoch+1}/{EPOCHS} - 验证损失: {val_loss:.4f}")

  # 早停逻辑与模型保存
    if val_loss < best_val_loss:
        best_val_loss = val_loss; torch.save(model.state_dict(), MODEL_SAVE_PATH)
        print(f"✅ 验证损失降低，最佳模型已保存至 {MODEL_SAVE_PATH}"); patience_counter = 0
    else:
        patience_counter += 1
        if patience_counter >= PATIENCE:
            print(f"触发早停。");
            break
```

### 定义模型(CRNN + ResNet)

```python
class ResNetBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super(ResNetBlock, self).__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=1, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels)

        self.shortcut = nn.Sequential()
        # 处理 stride 为 tuple 的情况
        if isinstance(stride, tuple):
            s_h, s_w = stride
            needs_conv = (s_h != 1 or s_w != 1) or (in_channels != out_channels)
        else:
            needs_conv = (stride != 1) or (in_channels != out_channels)

        if needs_conv:
            shortcut_stride = stride if isinstance(stride, tuple) else (stride, stride)
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=shortcut_stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        identity = self.shortcut(x)
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += identity
        return F.relu(out)


class CRNN_ResNet_Final(nn.Module):
    def __init__(self, vocab_size, img_height=32, img_width=256):
        super(CRNN_ResNet_Final, self).__init__()
        self.img_height = img_height
        self.img_width = img_width

        self.cnn = nn.Sequential(
            nn.Conv2d(1, 64, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(64), nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2),  # H:32→16, W:256→128

            ResNetBlock(64, 128, stride=(2, 1)),   # H:16→8,  W:128→128
            ResNetBlock(128, 256, stride=(2, 1)),  # H:8→4,   W:128→128
            ResNetBlock(256, 512, stride=(2, 2)),  # H:4→2,   W:128→64
        )

        # 自动推导 RNN 输入维度
        with torch.no_grad():
            dummy = torch.zeros(1, 1, img_height, img_width)
            cnn_out = self.cnn(dummy)
            _, C, H, W = cnn_out.shape
            self.cnn_out_channels = C
            self.cnn_out_height = H
            self.seq_len = W  # T = 64 for W=256

        rnn_input_size = C * H  # 512 * 2 = 1024 ← ✅ 修复点1
        self.rnn = nn.LSTM(rnn_input_size, 256, bidirectional=True, num_layers=2, batch_first=True, dropout=0.5)
        self.fc = nn.Linear(512, vocab_size)  # 256*2

    def get_seq_len(self, img_width=None):
        """计算 CTC 所需的序列长度 T"""
        if img_width is None:
            return self.seq_len
        # 动态计算：MaxPool(stride=2) + 最后一个 block width stride=2 → total /4
        return img_width // 4  # 256 → 64

    def forward(self, x):
        x = self.cnn(x)                     # [B, 512, 2, 64]
        x = x.permute(0, 3, 1, 2)           # [B, 64, 512, 2]
        x = x.flatten(2)                    # [B, 64, 1024] ← ✅ 维度匹配
        x, _ = self.rnn(x)                  # [B, 64, 512]
        x = self.fc(x)                      # [B, 64, vocab_size]
        return x.log_softmax(2)             # CTC 要求 log-prob
```

### 定义训练循环

```python
optimizer = optim.AdamW(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-4)
scheduler = optim.lr_scheduler.ReduceLROnPlateau(
    optimizer,
    mode='min',
    factor=0.5,
    patience=3,      # 微调数据少，patience 不宜过大
    min_lr=1e-7
)
criterion = nn.CTCLoss(blank=0, zero_infinity=True)

MODEL_SAVE_PATH = "best_crnn_finetune.pth"
best_val_loss = float('inf')
patience_counter = 0
PATIENCE = 15

def decode(preds):  # preds: [B, T, V]
    pred_indices = preds.argmax(-1)  # [B, T]
    texts = []
    for idxs in pred_indices:
        prev = "[BLANK]"
        decoded = []
        for i in idxs:
            char = int_to_char[i.item()]
            if char != "[BLANK]" and char != prev:
                decoded.append(char)
            prev = char
        texts.append("".join(decoded))
    return texts

print(f"🚀 开始训练/微调... (Epochs: {EPOCHS})")

for epoch in range(EPOCHS):
    model.train()
    train_loss = 0
    loop = tqdm(train_loader, desc=f"Ep {epoch+1}/{EPOCHS}")

    for imgs, labels, label_lens in loop:
        imgs, labels, label_lens = imgs.to(DEVICE), labels.to(DEVICE), label_lens.to(DEVICE)

        optimizer.zero_grad()
        seq_len = model.get_seq_len(IMAGE_WIDTH)  # = 64
        input_lens = torch.full((imgs.size(0),), seq_len, dtype=torch.long, device=DEVICE)
        preds = model(imgs).permute(1, 0, 2)  # [T, B, V]
        loss = criterion(preds, labels, input_lens, label_lens)
        loss.backward()
        optimizer.step()

        train_loss += loss.item()
        loop.set_postfix(loss=loss.item())

    # 验证
    model.eval()
    val_loss = 0
    with torch.no_grad():
        for i, (imgs, labels, label_lens) in enumerate(val_loader):
            imgs, labels, label_lens = imgs.to(DEVICE), labels.to(DEVICE), label_lens.to(DEVICE)
            seq_len = model.get_seq_len(IMAGE_WIDTH)  # = 64
            input_lens = torch.full((imgs.size(0),), seq_len, dtype=torch.long, device=DEVICE)
            preds = model(imgs).permute(1, 0, 2)  # [T, B, V]
            loss = criterion(preds, labels, input_lens, label_lens)
            val_loss += loss.item()

            # 打印几个示例看看效果
            if i == 0:
                pred_texts = decode(preds.permute(1,0,2))
                print(f"\n--- Epoch {epoch+1} 预测示例 ---")
                for j in range(min(5, len(pred_texts))):
                    real_text = "".join([int_to_char.get(c.item(), '')
                    for c in labels[j][:label_lens[j]]]);
                    print(f"  真实: {real_text:<20} -> 预测: {pred_texts[j]}")
                print("-" * 30)

    val_loss /= len(val_loader);
    print(f"Epoch {epoch+1}/{EPOCHS} - 验证损失: {val_loss:.4f}")

    scheduler.step(val_loss) # 更新学习率

    # === 手动打印当前学习率 ===
    current_lr = optimizer.param_groups[0]['lr']
    print(f"当前学习率: {current_lr:.2e}")

    if val_loss < best_val_loss:
        best_val_loss = val_loss
        torch.save({
            'model_state_dict': model.state_dict(),
            'optimizer_state_dict': optimizer.state_dict(),
            'best_val_loss': best_val_loss,
            'epoch': epoch,
        }, MODEL_SAVE_PATH)
        print("✅ 模型已保存")
        patience_counter = 0
    else:
        patience_counter += 1
        if patience_counter >= PATIENCE:
            print("🛑 早停触发")
            break
```

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

# --- 1. 重新定义模型类 (强制修正版) ---
# 我们在这里通过硬编码，确保它生成的壳子和你的 .pth 完美吻合

class ResNetBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super(ResNetBlock, self).__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=1, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels)

        self.shortcut = nn.Sequential()
        is_stride_tuple = isinstance(stride, tuple)
        if (is_stride_tuple and (stride[0]!=1 or stride[1]!=1)) or (not is_stride_tuple and stride!=1) or (in_channels != out_channels):
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        identity = self.shortcut(x)
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += identity
        return F.relu(out)

class CRNN_ResNet_Export(nn.Module):
    def __init__(self, vocab_size):
        super(CRNN_ResNet_Export, self).__init__()

        # CNN 结构保持不变
        self.cnn = nn.Sequential(
            nn.Conv2d(1, 64, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(64), nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2),
            ResNetBlock(64, 128, stride=(2, 1)),
            ResNetBlock(128, 256, stride=(2, 1)),
            ResNetBlock(256, 512, stride=(2, 2)),
        )

        # 【关键修改点】: 这里强制写死为 2048，不再让它自动计算，防止出错
        # 这就是为了迎合你已经训练好的 Checkpoint
        self.rnn_input_size = 2048

        self.rnn = nn.LSTM(self.rnn_input_size, 256, bidirectional=True, num_layers=2, batch_first=True, dropout=0.5)
        self.fc = nn.Linear(512, vocab_size)

    def forward(self, x):
        x = self.cnn(x)
        x = x.permute(0, 3, 1, 2)
        x = x.flatten(2) # 这里展平后就是 2048
        x, _ = self.rnn(x)
        x = self.fc(x)
        return x.log_softmax(2)
```
