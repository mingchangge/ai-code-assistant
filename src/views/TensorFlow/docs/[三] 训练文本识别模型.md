## 步骤 4: 定义PyTorch Dataset

```python
class OCRDataset(Dataset):
    def __init__(self, data_dir, transform=None):
        self.data_dir = data_dir
        self.transform = transform
        self.image_paths = []
        self.labels = []
        labels_path = os.path.join(data_dir, "labels.txt")
        with open(labels_path, "r", encoding="utf-8") as f:
            for line in f:
                path, label = line.strip().split("\t")
                self.image_paths.append(os.path.join(data_dir, path))
                self.labels.append(label)

    def __len__(self):
        return len(self.labels)

    def __getitem__(self, idx):
        img_path = self.image_paths[idx]
        label = self.labels[idx]
        image = Image.open(img_path).convert("L") # 直接转为灰度
        if self.transform:
            image = self.transform(image)

        label_tensor = torch.LongTensor([char_to_int[c] for c in label])
        return image, label_tensor
```

```python
EPOCHS = 200;
PATIENCE = 15;
best_val_loss = float('inf');
patience_counter = 0
MODEL_SAVE_PATH = "best_crnn_websafe_model.pth"

# --- 4. 【核心替换】定义全新的、Web友好的模型架构 ---
EPOCHS = 200
best_val_loss = float('inf')
patience_counter = 0
PATIENCE = 15 # 早停

# --- 1. 定义CRNN模型 ---
class ResNetBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super(ResNetBlock, self).__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.relu = nn.ReLU(inplace=True)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=1, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels)

        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += self.shortcut(x)
        out = self.relu(out)
        return out

class CRNN(nn.Module): # 我们保持类名不变，直接替换内容
    def __init__(self, vocab_size):
        super(CRNN, self).__init__()
        # ResNet-based CNN Backbone
        self.cnn = nn.Sequential(
            nn.Conv2d(1, 64, kernel_size=3, stride=1, padding=1, bias=False),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2), # 64 -> 32
            ResNetBlock(64, 128, stride=2),       # 32 -> 16
            ResNetBlock(128, 256, stride=2),      # 16 -> 8
            ResNetBlock(256, 512, stride=2),      # 8 -> 4
        )

        # RNN 部分
        self.rnn = nn.LSTM(512 * (IMAGE_HEIGHT // 16), 256, bidirectional=True, num_layers=2, batch_first=True, dropout=0.5)
        self.fc = nn.Linear(256 * 2, vocab_size)

    def forward(self, x):
        # CNN
        x = self.cnn(x) # [B, 512, H/16, W]
        x = x.permute(0, 3, 1, 2) # [B, W, C, H]
        x = x.reshape(x.size(0), x.size(1), -1) # [B, W, C*H]

        # RNN
        x, _ = self.rnn(x)

        # FC
        x = self.fc(x)
        x = x.log_softmax(2)
        return x

model = CRNN(VOCAB_SIZE).to(DEVICE)
criterion = nn.CTCLoss(blank=0, zero_infinity=True)
optimizer = optim.Adam(model.parameters(), lr=0.001)
# 添加一个学习率调度器，这里使用余弦退火
scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=EPOCHS, eta_min=0.00001)
```

## 步骤 8: 训练模型

```python
# 定义解码与可视化函数
def decode_predictions(preds, int_to_char):
    texts = []
    preds = preds.argmax(2)
    for pred in preds:
        text = ""
        for i, token_index in enumerate(pred):
            token = int_to_char[token_index.item()]
            if token != "[BLANK]" and (i == 0 or token != int_to_char[pred[i-1].item()]):
                text += token
        texts.append(text)
    return texts

# --- 3. 训练与验证循环 ---
print("🚀🚀🚀 开始PyTorch模型训练 🚀🚀🚀")

for epoch in range(EPOCHS):
    # 训练
    model.train()
    for images, labels, label_lengths in tqdm(train_loader, desc=f"Epoch {epoch+1}/{EPOCHS} [训练]"):
        images = images.to(DEVICE)
        labels = labels.to(DEVICE)
        label_lengths = label_lengths.to(DEVICE)

        optimizer.zero_grad()

        preds = model(images) # [B, T, C]
        preds = preds.permute(1, 0, 2) # [T, B, C] for CTCLoss
        input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)

        loss = criterion(preds, labels, input_lengths, label_lengths)
        loss.backward()
        optimizer.step()
        # 在每个epoch的训练循环结束后
        scheduler.step()

    # 验证与可视化
    model.eval()
    val_loss = 0
    with torch.no_grad():
        for i, (images, labels, label_lengths) in enumerate(val_loader):
            images = images.to(DEVICE)
            labels = labels.to(DEVICE)
            label_lengths = label_lengths.to(DEVICE)

            preds = model(images).permute(1, 0, 2)
            input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)

            loss = criterion(preds, labels, input_lengths, label_lengths)
            val_loss += loss.item()

            # 只在第一个验证批次进行可视化
            if i == 0:
                pred_texts = decode_predictions(preds.permute(1, 0, 2), int_to_char)
                print(f"\n--- Epoch {epoch+1} 预测示例 ---")
                for j in range(min(5, len(pred_texts))):
                    real_text = "".join([int_to_char[c.item()] for c in labels[j][:label_lengths[j]]])
                    print(f"  真实: {real_text} -> 预测: {pred_texts[j]}")
                print("-" * 30)

    val_loss /= len(val_loader)
    print(f"Epoch {epoch+1}/{EPOCHS} - 验证损失: {val_loss:.4f}")

    # 早停逻辑与模型保存
    if val_loss < best_val_loss:
        best_val_loss = val_loss
        torch.save(model.state_dict(), "best_crnn_model.pth")
        print("✅ 验证损失降低，模型已保存。")
        patience_counter = 0
    else:
        patience_counter += 1
        if patience_counter >= PATIENCE:
            print(f"验证损失连续{PATIENCE}个epoch未改善，触发早停。")
            break
```

  ```python
        self.bn2 = nn.BatchNorm2d(out_channels)

        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += self.shortcut(x)
        out = self.relu(out)
        return out

    class CRNN(nn.Module): # 我们保持类名不变，直接替换内容
        def __init__(self, vocab_size):
            super(CRNN, self).__init__()
            # ResNet-based CNN Backbone
            self.cnn = nn.Sequential(
                nn.Conv2d(1, 64, kernel_size=3, stride=1, padding=1, bias=False),
                nn.BatchNorm2d(64),
                nn.ReLU(inplace=True),
                nn.MaxPool2d(kernel_size=2, stride=2), # 64 -> 32
                ResNetBlock(64, 128, stride=2),       # 32 -> 16
                ResNetBlock(128, 256, stride=2),      # 16 -> 8
                ResNetBlock(256, 512, stride=2),      # 8 -> 4
            )

            # RNN 部分
            self.rnn = nn.LSTM(512 * (IMAGE_HEIGHT // 16), 256, bidirectional=True, num_layers=2, batch_first=True, dropout=0.5)
            self.fc = nn.Linear(256 * 2, vocab_size)

        def forward(self, x):
            # CNN
            x = self.cnn(x) # [B, 512, H/16, W]
            x = x.permute(0, 3, 1, 2) # [B, W, C, H]
            x = x.reshape(x.size(0), x.size(1), -1) # [B, W, C*H]

            # RNN
            x, _ = self.rnn(x)

            # FC
            x = self.fc(x)
            x = x.log_softmax(2)
            return x

  ```

  ```python
  class OCRDataset(Dataset):
    def __init__(self, data_dir, transform=None):
        self.data_dir = data_dir;
        self.transform = transform;
        self.image_paths = [];
        self.labels = []
        labels_path = os.path.join(data_dir, "labels.txt")
        with open(labels_path, "r", encoding="utf-8") as f:
            for line in f:
                path, label = line.strip().split("\t");
                self.image_paths.append(os.path.join(self.data_dir, path));
                self.labels.append(label)
    def __len__(self):
      return len(self.image_paths)
    def __getitem__(self, idx):
        img_path, label = self.image_paths[idx], self.labels[idx]
        image = Image.open(img_path).convert("L")
        if self.transform:
          image = self.transform(image)
        label_tensor = torch.LongTensor([char_to_int.get(c, 0) for c in label])
        return image, label_tensor

  IMAGE_HEIGHT = 64;
  IMAGE_WIDTH = 256;

  def collate_fn(batch):
    images, labels = zip(*batch); images = torch.stack(images, 0)
    label_lengths = torch.LongTensor([len(l) for l in labels])
    labels = nn.utils.rnn.pad_sequence(labels, batch_first=True, padding_value=0)
    return images, labels, label_lengths

  train_loader = DataLoader(train_dataset, batch_size=BATCH_SIZE, shuffle=True, collate_fn=collate_fn, num_workers=2)
  val_loader = DataLoader(val_dataset, batch_size=BATCH_SIZE, shuffle=False, collate_fn=collate_fn, num_workers=2)
  ```

  ```python
  def decode_predictions(preds, int_to_char_map):
      texts = []
      preds = preds.argmax(2)
      for pred in preds:
          text = ""
          for i, token_index in enumerate(pred):
              token = int_to_char[token_index.item()]
              if token != "[BLANK]" and (i == 0 or token != int_to_char[pred[i-1].item()]):
                  text += token
          texts.append(text)
      return texts

  print("\n🚀🚀🚀 开始在真实数据上进行微调...")
  for epoch in range(EPOCHS):
      model.train()
      train_loss = 0
      for images, labels, label_lengths in tqdm(train_loader, desc=f"Epoch {epoch+1}/{EPOCHS} [训练]"):
          images, labels, label_lengths = images.to(DEVICE), labels.to(DEVICE), label_lengths.to(DEVICE)

          optimizer.zero_grad();

          preds = model(images).permute(1, 0, 2)
          input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)
          loss = criterion(preds, labels, input_lengths, label_lengths)
          loss.backward()
          optimizer.step()
          train_loss += loss.item()

      scheduler.step() # 更新学习率

      model.eval()
      val_loss = 0
      with torch.no_grad():
          for i, (images, labels, label_lengths) in enumerate(val_loader):
              images, labels, label_lengths = images.to(DEVICE), labels.to(DEVICE), label_lengths.to(DEVICE)
              preds = model(images).permute(1, 0, 2)
              input_lengths = torch.full(size=(images.size(0),), fill_value=preds.size(0), dtype=torch.long)
              loss = criterion(preds, labels, input_lengths, label_lengths); val_loss += loss.item()
              if i == 0:
                  pred_texts = decode_predictions(preds.permute(1, 0, 2), int_to_char)
                  print(f"\n--- Epoch {epoch+1} 验证预测示例 ---")
                  for j in range(min(5, len(pred_texts))):
                      real_text = "".join([int_to_char.get(c.item(),'') for c in labels[j][:label_lengths[j]]]); print(f"  真实: {real_text:<20} -> 预测: {pred_texts[j]}")
                  print("-" * 30)

      avg_train_loss = train_loss / len(train_loader)
      avg_val_loss = val_loss / len(val_loader)
      print(f"Epoch {epoch+1}/{EPOCHS} - 训练损失: {avg_train_loss:.4f}, 验证损失: {avg_val_loss:.4f}, 学习率: {scheduler.get_last_lr()[0]:.6f}")

      if avg_val_loss < best_val_loss:
          best_val_loss = avg_val_loss
          torch.save(model.state_dict(), FINETUNED_MODEL_SAVE_PATH)
          print(f"✅ 验证损失降低，最佳模型已保存至 {FINETUNED_MODEL_SAVE_PATH}")
          patience_counter = 0
      else:
          patience_counter += 1
          print(f"验证损失未改善，Patience: {patience_counter}/{PATIENCE}")
          if patience_counter >= PATIENCE:
          print("触发早停。"); break
  ```
